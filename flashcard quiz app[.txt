  <script>
    const decks = {
      science: [
        {
          q: "What is photosynthesis?",
          a: "The process plants use to make food from sunlight."
        },
        {
          q: "What planet is known as the Red Planet?",
          a: "Mars, because iron minerals on its surface give it a reddish color."
        },
        {
          q: "What is the chemical formula for water?",
          a: "H₂O: two hydrogen atoms and one oxygen atom."
        },
        {
          q: "What is the center of an atom called?",
          a: "The nucleus."
        },
        {
          q: "What force pulls objects toward Earth?",
          a: "Gravity."
        }
      ],

      geography: [
        {
          q: "What is the largest ocean on Earth?",
          a: "The Pacific Ocean."
        },
        {
          q: "What is the capital of Japan?",
          a: "Tokyo."
        },
        {
          q: "On which continent is the Philippines?",
          a: "Asia."
        },
        {
          q: "What is the largest continent?",
          a: "Asia."
        },
        {
          q: "What do we call a large area of salt water?",
          a: "An ocean."
        }
      ],

      web: [
        {
          q: "What does HTML stand for?",
          a: "HyperText Markup Language."
        },
        {
          q: "What is CSS used for?",
          a: "CSS styles web pages, including colors, layouts, and fonts."
        },
        {
          q: "What does JavaScript do?",
          a: "It adds interactive behavior and functionality to web pages."
        },
        {
          q: "Which HTML tag creates a paragraph?",
          a: "The <p> tag."
        },
        {
          q: "Which HTML tag creates a hyperlink?",
          a: "The <a> tag."
        }
      ]
    };

    let currentDeck = "science";
    let cards = [...decks[currentDeck]];
    let index = 0;

    const ratings = {};

    const cardElement = document.getElementById("card");
    const questionElement = document.getElementById("question");
    const answerElement = document.getElementById("answer");
    const counterElement = document.getElementById("counter");
    const scoreElement = document.getElementById("score");
    const progressElement = document.getElementById("progress");

    function cardKey() {
      return currentDeck + ":" + index;
    }

    function renderCard() {
      const current = cards[index];

      questionElement.textContent = current.q;
      answerElement.textContent = current.a;

      cardElement.classList.remove("flipped");
      cardElement.setAttribute("aria-pressed", "false");

      counterElement.textContent =
        `Card ${index + 1} of ${cards.length}`;

      const masteredCount = cards.filter(
        (_, i) => ratings[currentDeck + ":" + i] === "known"
      ).length;

      scoreElement.textContent =
        `Mastered: ${masteredCount}`;

      progressElement.style.width =
        `${((index + 1) / cards.length) * 100}%`;
    }

    function flipCard() {
      cardElement.classList.toggle("flipped");

      cardElement.setAttribute(
        "aria-pressed",
        cardElement.classList.contains("flipped") ? "true" : "false"
      );
    }

    function nextCard() {
      index = (index + 1) % cards.length;
      renderCard();
    }

    function previousCard() {
      index = (index - 1 + cards.length) % cards.length;
      renderCard();
    }

    function shuffleCards() {
      for (let i = cards.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [cards[i], cards[j]] = [cards[j], cards[i]];
      }

      index = 0;
      renderCard();
    }

    function selectDeck(deckName) {
      currentDeck = deckName;
      cards = [...decks[deckName]];
      index = 0;

      document.querySelectorAll(".deck-btn").forEach(button => {
        button.classList.toggle(
          "active",
          button.dataset.deck === deckName
        );
      });

      renderCard();
    }

    function rateCard(rating) {
      ratings[cardKey()] = rating;
      renderCard();
      nextCard();
    }

    cardElement.addEventListener("click", flipCard);

    cardElement.addEventListener("keydown", event => {
      if (event.key === "Enter" || event.key === " ") {
        event.preventDefault();
        flipCard();
      }
    });

    document.getElementById("next")
      .addEventListener("click", nextCard);

    document.getElementById("prev")
      .addEventListener("click", previousCard);

    document.getElementById("shuffle")
      .addEventListener("click", shuffleCards);

    document.getElementById("learning")
      .addEventListener("click", () => rateCard("learning"));

    document.getElementById("mastered")
      .addEventListener("click", () => rateCard("known"));

    document.querySelectorAll(".deck-btn").forEach(button => {
      button.addEventListener("click", () => {
        selectDeck(button.dataset.deck);
      });
    });

    renderCard();
  </script>
